```python
import os
import socket
import subprocess
import time

import httpx
import pytest


def _find_free_port() -> int:
    with socket.socket() as s:
        s.bind(("127.0.0.1", 0))
        return s.getsockname()[1]


@pytest.fixture(scope="session")
def base_url():
    port = _find_free_port()
    env = os.environ.copy()
    env["STARLEARN_PORT"] = str(port)
    proc = subprocess.Popen(
        ["python", "start_server.py"],
        env=env,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )
    url = f"http://127.0.0.1:{port}"
    deadline = time.time() + 30
    while time.time() < deadline:
        try:
            r = httpx.get(f"{url}/api/health", timeout=2)
            if r.status_code == 200:
                break
        except Exception:
            pass
        time.sleep(0.5)
    else:
        proc.terminate()
        raise RuntimeError("Server failed to start in 30s")
    yield url
    proc.terminate()
    proc.wait(timeout=10)


@pytest.fixture
def guest_user(base_url):
    r = httpx.post(f"{base_url}/api/login/guest")
    assert r.status_code == 200, r.text
    return r.json()["user_id"]
```

```python
"""AuditAgent: 4 层防幻觉 + 越狱检测审核"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from app.services.tutor_engine.hallucination_guard import HallucinationGuard
from app.services.safety.jailbreak_detector import JailbreakDetector


@dataclass
class AuditResult:
    user_id: str
    risk_level: str  # "low" | "medium" | "high"
    blocked: bool
    reason: str
    jailbreak_score: float = 0.0
    hallucination_score: float = 0.0


class AuditAgent:
    """命名实体化的审核 Agent（L4 引用锚定 + 越狱检测）"""

    name = "audit_agent"

    def __init__(
        self,
        guard: HallucinationGuard | None = None,
        jailbreak: JailbreakDetector | None = None,
    ) -> None:
        self._guard = guard or HallucinationGuard()
        self._jailbreak = jailbreak or JailbreakDetector()

    async def run(
        self,
        user_id: str,
        input_text: str,
        output_text: str,
        knowledge_source: list[str] | None = None,
    ) -> AuditResult:
        # L0 越狱检测
        jb_result = await self._jailbreak.scan(input_text)
        if jb_result.risk_score > 0.8:
            return AuditResult(
                user_id=user_id,
                risk_level="high",
                blocked=True,
                reason=f"jailbreak: {jb_result.pattern}",
                jailbreak_score=jb_result.risk_score,
            )

        # L1-L4 防幻觉
        guard_result = await self._guard.check(
            input_text=input_text,
            output_text=output_text,
            knowledge_source=knowledge_source or [],
        )

        if not guard_result.passed:
            return AuditResult(
                user_id=user_id,
                risk_level=guard_result.risk_level,
                blocked=True,
                reason=guard_result.reason,
                hallucination_score=guard_result.score,
            )

        return AuditResult(
            user_id=user_id,
            risk_level="low",
            blocked=False,
            reason="all checks passed",
            jailbreak_score=jb_result.risk_score,
            hallucination_score=guard_result.score,
        )
```

```python
# app/services/orchestrator/chain.py
"""OrchestratorChain: 任意 Agent → 任意 Agent 的链式重试 + 共识投票"""
from __future__ import annotations

import asyncio
import logging
from dataclasses import dataclass, field
from typing import Any, Awaitable, Callable

logger = logging.getLogger("starlearn.orchestrator.chain")

ChainFn = Callable[[dict[str, Any]], Awaitable[dict[str, Any]]]


@dataclass
class ChainLink:
    name: str
    fn: ChainFn
    max_retries: int = 0
    backoff_seconds: float = 0.5


@dataclass
class ChainResult:
    success: bool
    state: dict[str, Any]
    errors: list[str] = field(default_factory=list)
    links_executed: list[str] = field(default_factory=list)


class OrchestratorChain:
    def __init__(self, links: list[ChainLink]) -> None:
        if not links:
            raise ValueError("OrchestratorChain requires at least one link")
        self._links = links

    async def run(self, initial_state: dict[str, Any]) -> ChainResult:
        state = dict(initial_state)
        errors: list[str] = []
        executed: list[str] = []

        for link in self._links:
            attempt = 0
            while attempt <= link.max_retries:
                try:
                    state = await link.fn(state)
                    executed.append(link.name)
                    break
                except Exception as exc:  # noqa: BLE001
                    attempt += 1
                    if attempt > link.max_retries:
                        msg = f"{link.name} failed after {attempt - 1} retries: {exc}"
                        logger.error(msg)
                        errors.append(msg)
                        return ChainResult(success=False, state=state, errors=errors, links_executed=executed)
                    await asyncio.sleep(link.backoff_seconds * attempt)

        return ChainResult(success=True, state=state, errors=errors, links_executed=executed)
```

```python
# app/services/safety/jailbreak_detector.py
"""越狱检测：L0 正则（<50ms）+ L1 LLM 二次判定（<2s）"""
from __future__ import annotations

import re
from dataclasses import dataclass


@dataclass
class JailbreakResult:
    risk_score: float  # 0.0-1.0
    pattern: str  # 命中的模式名
    matched_text: str = ""


class JailbreakDetector:
    # L0 正则模式（覆盖常见越狱 payload）
    L0_PATTERNS = [
        ("ignore_previous", re.compile(r"ignore\s+(previous|all|above)\s+(instructions?|prompts?)", re.I)),
        ("role_escape", re.compile(r"you\s+are\s+now\s+(DAN|jailbreak|evil|admin)", re.I)),
        ("system_prompt_leak", re.compile(r"(reveal|show|print)\s+(your\s+)?(system\s+prompt|instructions)", re.I)),
        ("prompt_injection", re.compile(r"<\|im_start\|>|<\|im_end\|>|\[INST\]|\[/INST\]", re.I)),
        ("bypass_safety", re.compile(r"bypass\s+(safety|filter|restriction)", re.I)),
    ]

    def __init__(self, level: str = "L0") -> None:
        if level not in ("L0", "L1", "L0+L1"):
            raise ValueError(f"invalid level: {level}")
        self.level = level

    async def scan(self, text: str) -> JailbreakResult:
        # L0 正则检测
        max_score = 0.0
        matched_pattern = ""
        matched_text = ""
        for pattern_name, regex in self.L0_PATTERNS:
            m = regex.search(text)
            if m:
                score = 0.85
                if max_score < score:
                    max_score = score
                    matched_pattern = pattern_name
                    matched_text = m.group(0)

        # L1 LLM 二次判定（异步，可选）
        if self.level in ("L1", "L0+L1") and max_score < 0.5:
            llm_score = await self._llm_check(text)
            if llm_score > max_score:
                max_score = llm_score
                matched_pattern = "llm_judge"

        return JailbreakResult(risk_score=max_score, pattern=matched_pattern, matched_text=matched_text)

    async def _llm_check(self, text: str) -> float:
        """L1 LLM 二次判定（默认禁用，需要 ENABLE_L1_JAILBREAK=true）"""
        # 占位：调用 llm_stream.call_llm_async_messages
        # 简化版：直接返回 0（避免增加 M3 测试负担）
        return 0.0
```

```python
# app/services/learning_path/forgetting_curve.py
"""SM-2 间隔重复算法（Anki 经典实现 + 个性化 stability 拟合）"""
from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime, timedelta


@dataclass
class LearningState:
    user_id: str
    topic_id: str
    stability: float = 2.5  # 个性化遗忘率参数（基于历史拟合）
    difficulty: float = 5.0  # 0-10 难度
    reps: int = 0  # 连续正确次数
    lapses: int = 0  # 失败次数
    last_review: datetime | None = None
    interval: float = 1.0  # 当前间隔（天）


class SM2Scheduler:
    def next_review(self, state: LearningState, now: datetime) -> datetime:
        """根据 state 计算下次复习时间"""
        return now + timedelta(days=state.interval)

    def review(self, state: LearningState, quality: int, now: datetime) -> tuple[LearningState, datetime]:
        """根据答题质量（0-5）更新 state"""
        new_state = LearningState(
            user_id=state.user_id,
            topic_id=state.topic_id,
            stability=state.stability,
            difficulty=state.difficulty,
            reps=state.reps,
            lapses=state.lapses,
            last_review=now,
            interval=state.interval,
        )

        if quality < 3:
            # 失败：重置 reps，lapses + 1
            new_state.reps = 0
            new_state.lapses += 1
            new_state.interval = 1.0
        else:
            # 成功：根据 SM-2 公式更新 interval
            new_state.reps += 1
            if new_state.reps == 1:
                new_state.interval = 1.0
            elif new_state.reps == 2:
                new_state.interval = 3.0
            else:
                new_state.interval = state.interval * state.stability

        # 更新 stability（基于答题质量）
        new_state.stability = max(1.3, state.stability + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02)))

        return new_state, self.next_review(new_state, now)
```

```python
"""每日复习提醒调度器"""
from __future__ import annotations

import logging
from collections import defaultdict
from datetime import datetime

from app.services.learning_path.forgetting_curve import LearningState, SM2Scheduler

logger = logging.getLogger("starlearn.review_scheduler")


class ReviewScheduler:
    def __init__(self, max_per_user_per_day: int = 2) -> None:
        self.max_per_user_per_day = max_per_user_per_day
        self._sm2 = SM2Scheduler()

    async def run_daily_review(self, now: datetime | None = None) -> int:
        now = now or datetime.utcnow()
        states = await self._fetch_due_states(now)

        # 按用户分组，每个用户最多 max_per_user_per_day
        per_user: dict[str, list[LearningState]] = defaultdict(list)
        for s in states:
            if len(per_user[s.user_id]) < self.max_per_user_per_day:
                per_user[s.user_id].append(s)

        sent = 0
        for user_id, user_states in per_user.items():
            for state in user_states:
                await self._send_review_notification(user_id=user_id, topic_id=state.topic_id)
                sent += 1

        logger.info(f"Daily review sent {sent} notifications to {len(per_user)} users")
        return sent

    async def _fetch_due_states(self, now: datetime) -> list[LearningState]:
        """占位：从 DB 拉取 next_review <= now 的状态"""
        return []

    async def _send_review_notification(self, user_id: str, topic_id: str) -> None:
        """占位：发送复习提醒（push / email）"""
        logger.info(f"Review notification: user={user_id}, topic={topic_id}")
```

```python
# app/agents/critic.py
"""CriticAgent: L3 防幻觉独立 Agent（专门挑错）"""
from __future__ import annotations

from dataclasses import dataclass


@dataclass
class CritiqueResult:
    quality: str  # "low" | "medium" | "high"
    score: float  # 0.0-1.0
    issues: list[str]


class CriticAgent:
    name = "critic_agent"

    async def review(self, answer: str, reference: str) -> CritiqueResult:
        """简化版：基于关键词匹配 + 长度评分"""
        score = self._score(answer, reference)
        if score > 0.8:
            quality = "high"
        elif score > 0.5:
            quality = "medium"
        else:
            quality = "low"

        issues = []
        if len(answer) < 10:
            issues.append("answer_too_short")
        if reference and not any(word in answer for word in reference.split()[:5]):
            issues.append("missing_key_terms")

        return CritiqueResult(quality=quality, score=score, issues=issues)

    def _score(self, answer: str, reference: str) -> float:
        if not reference:
            return 0.5
        ref_words = set(reference.lower().split())
        ans_words = set(answer.lower().split())
        overlap = len(ref_words & ans_words)
        return min(1.0, overlap / max(1, len(ref_words)))
```

```python
# app/services/sandbox/executor.py
"""Sandbox: subprocess + timeout 执行 Python 代码（不引入 Docker 依赖）"""
from __future__ import annotations

import asyncio
import logging
from dataclasses import dataclass

logger = logging.getLogger("starlearn.sandbox")

BLOCKED_MODULES = {"os", "subprocess", "sys", "shutil", "socket", "ctypes"}


@dataclass
class ExecutionResult:
    success: bool
    stdout: str
    stderr: str
    error: str = ""


class SandboxExecutor:
    def __init__(self, timeout_seconds: int = 5) -> None:
        self.timeout_seconds = timeout_seconds

    async def run_python(self, code: str) -> ExecutionResult:
        # 前置安全检查：拦截危险 import
        for mod in BLOCKED_MODULES:
            if f"import {mod}" in code or f"from {mod}" in code:
                return ExecutionResult(
                    success=False, stdout="", stderr="",
                    error=f"blocked: import {mod} not allowed",
                )

        try:
            proc = await asyncio.create_subprocess_exec(
                "python", "-c", code,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            stdout, stderr = await asyncio.wait_for(
                proc.communicate(), timeout=self.timeout_seconds,
            )
            return ExecutionResult(
                success=proc.returncode == 0,
                stdout=stdout.decode("utf-8", errors="ignore"),
                stderr=stderr.decode("utf-8", errors="ignore"),
            )
        except asyncio.TimeoutError:
            try:
                proc.kill()
            except Exception:  # noqa: BLE001
                pass
            return ExecutionResult(success=False, stdout="", stderr="", error=f"timeout after {self.timeout_seconds}s")
        except Exception as exc:  # noqa: BLE001
            return ExecutionResult(success=False, stdout="", stderr="", error=str(exc))
```

```python
"""认知风格识别（视觉/听觉/动觉）—— 基于行为聚类"""
from __future__ import annotations


class StyleRecognizer:
    CHANNELS = ["visual", "auditory", "kinesthetic"]

    def classify(self, behavior: dict) -> dict:
        """根据页面停留时长判断主要风格"""
        scores = {
            "visual": behavior.get("image_page_avg_dwell_seconds", 0),
            "auditory": behavior.get("audio_page_avg_dwell_seconds", 0),
            "kinesthetic": behavior.get("code_editor_avg_dwell_seconds", 0),
        }
        primary = max(scores, key=scores.get)
        total = sum(scores.values()) or 1
        confidence = scores[primary] / total

        return {
            "primary": primary,
            "confidence": round(confidence, 2),
            "scores": scores,
        }
```

```python
"""变式题生成器：同知识点不同情境/表述"""
from __future__ import annotations

import copy

SCENARIOS = ["水果", "动物", "交通工具", "文具", "食物"]


class VariantGenerator:
    def generate(self, base: dict, count: int = 3) -> list[dict]:
        variants = []
        for i in range(count):
            v = copy.deepcopy(base)
            # 换情境
            v["scenario"] = SCENARIOS[(SCENARIOS.index(base.get("scenario", "水果")) + i + 1) % len(SCENARIOS)]
            # 换表述（简化版：在 stem 前加情境前缀）
            v["stem"] = f"如果用 {v['scenario']} 来算：{base['stem']}"
            v["is_variant"] = True
            v["parent_id"] = base.get("id")
            v["id"] = f"{base.get('id', 'ex')}_v{i+1}"
            variants.append(v)
        return variants
```

```python
"""反思日志 Agent：每次学习结束弹出 3 个元认知问题"""
from __future__ import annotations

from collections import defaultdict


class ReflectionLogAgent:
    QUESTIONS = [
        "刚才卡在哪一步？为什么？",
        "如果换一种条件，你会怎么做？",
        "怎么用自己的话给别人讲一遍？",
    ]

    async def generate_questions(self, topic: str) -> list[str]:
        return [q + f"（针对：{topic}）" for q in self.QUESTIONS]

    async def aggregate_weekly(self, reflections: list[dict]) -> dict:
        grouped: dict[str, list[dict]] = defaultdict(list)
        for r in reflections:
            grouped[r["topic"]].append(r)
        return {
            topic: {"count": len(items), "samples": [i["answer"] for i in items[:3]]}
            for topic, items in grouped.items()
        }
```

```python
"""教研知识自动沉淀：教师编辑 → 审核 → 入私有 KB"""
from __future__ import annotations

import logging
import uuid

logger = logging.getLogger("starlearn.kb.deposition")


class KnowledgeDepositionService:
    def __init__(self) -> None:
        self._reviews: dict[str, dict] = {}

    async def submit_for_review(self, teacher_id: str, content: str, knowledge_point: str) -> dict:
        review_id = f"rv_{uuid.uuid4().hex[:8]}"
        self._reviews[review_id] = {
            "teacher_id": teacher_id,
            "content": content,
            "knowledge_point": knowledge_point,
            "status": "pending_review",
        }
        logger.info(f"KB deposition submitted: {review_id}")
        return {"review_id": review_id, "status": "pending_review"}

    async def approve_review(self, review_id: str, approver: str) -> dict:
        review = self._reviews.get(review_id)
        if not review:
            return {"status": "not_found"}
        review["status"] = "approved"
        review["approver"] = approver
        review["entered_kb"] = True
        # 占位：实际写入私有 KB
        return {"status": "approved", "entered_kb": True}

    async def _approve(self, review_id: str) -> bool:
        return True
```

```python
"""多模态输入理解：OCR + 解题步骤逻辑分析"""
from __future__ import annotations

import base64
import logging

logger = logging.getLogger("starlearn.multimodal.ocr")


class MultimodalOCR:
    async def extract_text(self, image_base64: str) -> str:
        """简化版：base64 解码后返回占位文本（生产接 Seedream/MiniMax-Vision）"""
        try:
            base64.b64decode(image_base64)
        except Exception as exc:  # noqa: BLE001
            logger.warning(f"Invalid base64 image: {exc}")
            return ""
        return "[OCR placeholder text]"

    async def analyze_solution(self, text: str, knowledge_point: str) -> dict:
        """简化版：判断解题步骤是否有逻辑断裂"""
        lines = [l.strip() for l in text.split("\n") if l.strip()]
        return {
            "verdict": "正确" if "=" in text and len(lines) >= 2 else "需检查",
            "step_count": len(lines),
            "logic_break": None if len(lines) >= 2 else "步骤过少",
            "knowledge_point": knowledge_point,
        }
```
